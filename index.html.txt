const GAME_CONFIG = {
    fieldWidth: 860,
    fieldHeight: 480,
    midlineX: 430,
    trainingDuration: 60,
    combatDuration: 240,
};

let gameState = {
    phase: 'TRAINING',
    timer: GAME_CONFIG.trainingDuration,
    qrOdd: Math.random() > 0.5,
    figureTarget: ['Walker', 'Yoyo', 'Walker X'][Math.floor(Math.random() * 3)],
    winner: null,
    winReason: ''
};

class Team {
    constructor(id, name, color, isAi = false) {
        this.id = id;
        this.name = name;
        this.color = color;
        this.isAi = isAi;
        this.score = 0;
        this.trainingScore = 0;
        this.ballsInStation = 0;
        this.transferBalls = 0;
        this.yellowCards = 0;
        this.figureIdentified = false;
        this.qrIdentified = false;
        this.towerRingKnocked = false;
        this.flagUnlocked = false;
        this.flagKnocked = false;
        this.blocksStacked = 0;
        this.requiredBalls = 8;
    }

    calculateTotalScore() {
        let combatBallScore = this.ballsInStation * 10;
        if (this.blocksStacked > 0) combatBallScore *= 2;
        let ringScore = this.towerRingKnocked ? 50 : 0;
        let penalties = this.yellowCards * 10;
        this.score = Math.max(0, this.trainingScore + combatBallScore + ringScore - penalties);
        return this.score;
    }
}

const redTeam = new Team('red', '紅方', '#ef4444', false);
const blueTeam = new Team('blue', '藍方', '#3b82f6', true);

class Robot {
    constructor(x, y, team, type = 'engineer') {
        this.x = x;
        this.y = y;
        this.team = team;
        this.type = type;
        this.radius = 18;
        this.speed = type === 'transform' ? 3.3 : 2.5;
        this.angle = team === 'red' ? 0 : Math.PI;
        this.vx = 0;
        this.vy = 0;
        this.charge = 0;
        this.maxCharge = 50;
    }

    update() {
        this.x += this.vx;
        this.y += this.vy;

        const minX = this.team === 'red' ? this.radius + 10 : GAME_CONFIG.midlineX + this.radius + 5;
        const maxX = this.team === 'red' ? GAME_CONFIG.midlineX - this.radius - 5 : GAME_CONFIG.fieldWidth - this.radius - 10;
        const minY = this.radius + 10;
        const maxY = GAME_CONFIG.fieldHeight - this.radius - 10;

        if (this.x < minX) this.x = minX;
        if (this.x > maxX) this.x = maxX;
        if (this.y < minY) this.y = minY;
        if (this.y > maxY) this.y = maxY;
    }

    draw() {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.angle);

        ctx.fillStyle = this.team === 'red' ? '#dc2626' : '#2563eb';
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.roundRect(-this.radius, -this.radius, this.radius * 2, this.radius * 2, 4);
        ctx.fill();
        ctx.stroke();

        ctx.fillStyle = '#64748b';
        if (this.type === 'engineer') {
            ctx.fillRect(this.radius - 2, -6, 10, 3);
            ctx.fillRect(this.radius - 2, 3, 10, 3);
        } else {
            ctx.fillRect(this.radius - 2, -4, 12, 8);
        }

        // 左指示燈 (QR ID: 奇紅偶藍)
        ctx.fillStyle = gameState.qrOdd ? '#ef4444' : '#3b82f6';
        ctx.beginPath();
        ctx.arc(0, -this.radius + 4, 3, 0, Math.PI * 2);
        ctx.fill();

        // 右指示燈 (公仔)
        let fCol = '#ef4444';
        if (gameState.figureTarget === 'Yoyo') fCol = '#3b82f6';
        if (gameState.figureTarget === 'Walker X') fCol = '#a855f7';
        ctx.fillStyle = fCol;
        ctx.beginPath();
        ctx.arc(0, this.radius - 4, 3, 0, Math.PI * 2);
        ctx.fill();

        // 蓄力條
        if (this.charge > 0) {
            ctx.fillStyle = '#facc15';
            ctx.fillRect(-this.radius, -this.radius - 8, (this.radius * 2) * (this.charge / this.maxCharge), 4);
        }

        ctx.restore();
    }
}

const redRobots = [
    new Robot(80, 140, 'red', 'engineer'),
    new Robot(80, 340, 'red', 'transform')
];
let activeRedIndex = 0;

const blueRobots = [
    new Robot(780, 140, 'blue', 'engineer'),
    new Robot(780, 340, 'blue', 'transform')
];

let balls = [];
class Ball {
    constructor(x, y, vx, vy, color, fromTeam) {
        this.x = x;
        this.y = y;
        this.vx = vx;
        this.vy = vy;
        this.color = color;
        this.fromTeam = fromTeam;
        this.radius = 6;
        this.active = true;
    }

    update() {
        this.x += this.vx;
        this.y += this.vy;
        this.vx *= 0.96;
        this.vy *= 0.96;

        if (this.x < 10 || this.x > GAME_CONFIG.fieldWidth - 10) this.vx *= -0.7;
        if (this.y < 10 || this.y > GAME_CONFIG.fieldHeight - 10) this.vy *= -0.7;

        checkBallScoring(this);
    }

    draw() {
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fillStyle = this.color;
        ctx.strokeStyle = '#000';
        ctx.lineWidth = 1;
        ctx.fill();
        ctx.stroke();
    }
}

const ARENA = {
    redStation: { x: 380, y: 190, w: 45, h: 100 },
    blueStation: { x: 435, y: 190, w: 45, h: 100 },
    redTower: { x: 260, y: 80, radius: 16, knocked: false },
    blueTower: { x: 600, y: 80, radius: 16, knocked: false },
    redFlag: { x: 120, y: 240, w: 12, h: 36, knocked: false },
    blueFlag: { x: 740, y: 240, w: 12, h: 36, knocked: false },
    transferZoneRed: { x: 200, y: 400, w: 80, h: 70 },
    transferZoneBlue: { x: 580, y: 400, w: 80, h: 70 },
};

function checkBallScoring(ball) {
    if (!ball.active) return;
    // 計分站判定
    if (ball.x > ARENA.redStation.x && ball.x < ARENA.redStation.x + ARENA.redStation.w &&
        ball.y > ARENA.redStation.y && ball.y < ARENA.redStation.y + ARENA.redStation.h) {
        ball.active = false;
        redTeam.ballsInStation++;
        checkFlagUnlock(redTeam);
    }
    if (ball.x > ARENA.blueStation.x && ball.x < ARENA.blueStation.x + ARENA.blueStation.w &&
        ball.y > ARENA.blueStation.y && ball.y < ARENA.blueStation.y + ARENA.blueStation.h) {
        ball.active = false;
        blueTeam.ballsInStation++;
        checkFlagUnlock(blueTeam);
    }

    // 擊倒前哨塔環判定 (50分)
    if (ball.fromTeam === 'red' && !ARENA.blueTower.knocked) {
        let dist = Math.hypot(ball.x - ARENA.blueTower.x, ball.y - ARENA.blueTower.y);
        if (dist < ARENA.blueTower.radius + ball.radius) {
            ARENA.blueTower.knocked = true;
            redTeam.towerRingKnocked = true;
            ball.active = false;
            checkFlagUnlock(redTeam);
        }
    }
    if (ball.fromTeam === 'blue' && !ARENA.redTower.knocked) {
        let dist = Math.hypot(ball.x - ARENA.redTower.x, ball.y - ARENA.redTower.y);
        if (dist < ARENA.redTower.radius + ball.radius) {
            ARENA.redTower.knocked = true;
            blueTeam.towerRingKnocked = true;
            ball.active = false;
            checkFlagUnlock(blueTeam);
        }
    }

    // 擊倒軍旗判定 (直接絕殺勝)
    if (ball.fromTeam === 'red' && redTeam.flagUnlocked && !ARENA.blueFlag.knocked) {
        if (ball.x > ARENA.blueFlag.x && ball.x < ARENA.blueFlag.x + ARENA.blueFlag.w &&
            ball.y > ARENA.blueFlag.y && ball.y < ARENA.blueFlag.y + ARENA.blueFlag.h) {
            ARENA.blueFlag.knocked = true;
            triggerDirectVictory(redTeam, '紅方小球擊倒對方軍旗，達成【絕殺勝利】！');
        }
    }
    if (ball.fromTeam === 'blue' && blueTeam.flagUnlocked && !ARENA.redFlag.knocked) {
        if (ball.x > ARENA.redFlag.x && ball.x < ARENA.redFlag.x + ARENA.redFlag.w &&
            ball.y > ARENA.redFlag.y && ball.y < ARENA.redFlag.y + ARENA.redFlag.h) {
            ARENA.redFlag.knocked = true;
            triggerDirectVictory(blueTeam, '藍方小球擊倒對方軍旗，達成【絕殺勝利】！');
        }
    }
}

function checkFlagUnlock(team) {
    if (team.ballsInStation >= team.requiredBalls && team.towerRingKnocked && team.yellowCards < 3) {
        team.flagUnlocked = true;
    }
}

function triggerDirectVictory(team, reason) {
    gameState.phase = 'ENDED';
    gameState.winner = team.name;
    gameState.winReason = reason;
}

function setupTouchEvents(id, keyName) {
    const el = document.getElementById(id);
    if (!el) return;
    el.addEventListener('touchstart', (e) => { e.preventDefault(); keys[keyName] = true; });
    el.addEventListener('touchend', (e) => { e.preventDefault(); keys[keyName] = false; });
    el.addEventListener('mousedown', (e) => { keys[keyName] = true; });
    el.addEventListener('mouseup', (e) => { keys[keyName] = false; });
}

function shootBall(robot) {
    if (gameState.phase !== 'COMBAT') return;
    const force = Math.max(5, (robot.charge / robot.maxCharge) * 14);
    robot.charge = 0;
    const vx = Math.cos(robot.angle) * force;
    const vy = Math.sin(robot.angle) * force;
    const color = Math.random() > 0.5 ? '#f97316' : '#a855f7';
    balls.push(new Ball(robot.x + Math.cos(robot.angle) * 22, robot.y + Math.sin(robot.angle) * 22, vx, vy, color, robot.team));
}

// 比賽定時器
setInterval(() => {
    if (gameState.phase === 'ENDED') return;

    gameState.timer--;
    if (gameState.timer <= 0) {
        if (gameState.phase === 'TRAINING') {
            concludeTrainingPhase();
            gameState.phase = 'COMBAT';
            gameState.timer = GAME_CONFIG.combatDuration;
            document.getElementById('phase-name').innerText = '對戰階段 (手動操作)';
        } else if (gameState.phase === 'COMBAT') {
            gameState.phase = 'ENDED';
            redTeam.calculateTotalScore();
            blueTeam.calculateTotalScore();
            if (redTeam.score > blueTeam.score) {
                gameState.winner = redTeam.name;
                gameState.winReason = `時間結束，積分勝出 (${redTeam.score} : ${blueTeam.score})`;
            } else if (blueTeam.score > redTeam.score) {
                gameState.winner = blueTeam.name;
                gameState.winReason = `時間結束，積分勝出 (${blueTeam.score} : ${redTeam.score})`;
            } else {
                gameState.winner = '平局';
                gameState.winReason = `雙方總積分相同 (${redTeam.score} : ${blueTeam.score})`;
            }
        }
    }
    document.getElementById('timer-val').innerText = gameState.timer;
}, 1000);

function concludeTrainingPhase() {
    redTeam.transferBalls = Math.floor(Math.random() * 3) + 2;
    blueTeam.transferBalls = Math.floor(Math.random() * 3) + 2;
    redTeam.trainingScore += redTeam.transferBalls * 10;
    blueTeam.trainingScore += blueTeam.transferBalls * 10;
    redTeam.ballsInStation += redTeam.transferBalls;
    blueTeam.ballsInStation += blueTeam.transferBalls;

    redTeam.trainingScore += 40;
    blueTeam.trainingScore += 40;
    redTeam.qrIdentified = true;
    blueTeam.qrIdentified = true;

    redTeam.figureIdentified = true;
    redTeam.trainingScore += 20;
    redTeam.requiredBalls -= 2;

    blueTeam.figureIdentified = true;
    blueTeam.trainingScore += 20;
    blueTeam.requiredBalls -= 2;

    redTeam.trainingScore += 100;
    blueTeam.trainingScore += 100;
}

function update() {
    if (gameState.phase === 'COMBAT') {
        const curRobot = redRobots[activeRedIndex];
        curRobot.vx = 0;
        curRobot.vy = 0;
        if (keys['KeyW'] || keys['ArrowUp']) curRobot.vy = -curRobot.speed;
        if (keys['KeyS'] || keys['ArrowDown']) curRobot.vy = curRobot.speed;
        if (keys['KeyA'] || keys['ArrowLeft']) curRobot.vx = -curRobot.speed;
        if (keys['KeyD'] || keys['ArrowRight']) curRobot.vx = curRobot.speed;

        if (curRobot.vx !== 0 || curRobot.vy !== 0) {
            curRobot.angle = Math.atan2(curRobot.vy, curRobot.vx);
        }

        if (keys['Space']) {
            if (curRobot.charge < curRobot.maxCharge) curRobot.charge++;
        }

        // AI 藍方行動
        const aiBot = blueRobots[1];
        if (!blueTeam.towerRingKnocked) {
            const angleToTower = Math.atan2(ARENA.redTower.y - aiBot.y, ARENA.redTower.x - aiBot.x);
            aiBot.angle = angleToTower;
            if (Math.random() < 0.02) {
                balls.push(new Ball(aiBot.x, aiBot.y, Math.cos(angleToTower) * 9, Math.sin(angleToTower) * 9, '#f97316', 'blue'));
            }
        } else if (blueTeam.flagUnlocked && !ARENA.redFlag.knocked) {
            const angleToFlag = Math.atan2(ARENA.redFlag.y - aiBot.y, ARENA.redFlag.x - aiBot.x);
            aiBot.angle = angleToFlag;
            if (Math.random() < 0.03) {
                balls.push(new Ball(aiBot.x, aiBot.y, Math.cos(angleToFlag) * 11, Math.sin(angleToFlag) * 11, '#a855f7', 'blue'));
            }
        } else {
            const angleToStation = Math.atan2(ARENA.blueStation.y + 40 - aiBot.y, ARENA.blueStation.x - aiBot.x);
            aiBot.angle = angleToStation;
            if (Math.random() < 0.025) {
                balls.push(new Ball(aiBot.x, aiBot.y, Math.cos(angleToStation) * 8, Math.sin(angleToStation) * 8, '#f97316', 'blue'));
            }
        }
    }

    redRobots.forEach(r => r.update());
    blueRobots.forEach(r => r.update());
    balls.forEach(b => b.update());
    balls = balls.filter(b => b.active);

    redTeam.calculateTotalScore();
    blueTeam.calculateTotalScore();
    document.getElementById('red-score').innerText = redTeam.score;
    document.getElementById('blue-score').innerText = blueTeam.score;
    document.getElementById('red-station-balls').innerText = redTeam.ballsInStation;
    document.getElementById('blue-station-balls').innerText = blueTeam.ballsInStation;
}

function drawField() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = '#0b1329';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.strokeStyle = '#334155';
    ctx.strokeRect(10, 10, canvas.width - 20, canvas.height - 20);

    ctx.fillStyle = 'rgba(234, 88, 12, 0.2)';
    ctx.fillRect(ARENA.transferZoneRed.x, ARENA.transferZoneRed.y, ARENA.transferZoneRed.w, ARENA.transferZoneRed.h);
    ctx.fillStyle = 'rgba(59, 130, 246, 0.2)';
    ctx.fillRect(ARENA.transferZoneBlue.x, ARENA.transferZoneBlue.y, ARENA.transferZoneBlue.w, ARENA.transferZoneBlue.h);

    // 中線 (Midline)
    ctx.strokeStyle = '#eab308';
    ctx.lineWidth = 4;
    ctx.setLineDash([8, 6]);
    ctx.beginPath();
    ctx.moveTo(GAME_CONFIG.midlineX, 10);
    ctx.lineTo(GAME_CONFIG.midlineX, canvas.height - 10);
    ctx.stroke();
    ctx.setLineDash([]);

    // 球站
    ctx.fillStyle = '#991b1b';
    ctx.fillRect(ARENA.redStation.x, ARENA.redStation.y, ARENA.redStation.w, ARENA.redStation.h);
    ctx.fillStyle = '#1e40af';
    ctx.fillRect(ARENA.blueStation.x, ARENA.blueStation.y, ARENA.blueStation.w, ARENA.blueStation.h);
    ctx.fillStyle = '#fff';
    ctx.font = '10px sans-serif';
    ctx.fillText('紅球站', ARENA.redStation.x + 6, ARENA.redStation.y + 50);
    ctx.fillText('藍球站', ARENA.blueStation.x + 6, ARENA.blueStation.y + 50);

    // 前哨塔環
    drawTower(ARENA.redTower, redTeam.towerRingKnocked, '#ef4444');
    drawTower(ARENA.blueTower, blueTeam.towerRingKnocked, '#3b82f6');

    // 軍旗
    drawFlag(ARENA.redFlag, redTeam.flagUnlocked, ARENA.redFlag.knocked, '#ef4444');
    drawFlag(ARENA.blueFlag, blueTeam.flagUnlocked, ARENA.blueFlag.knocked, '#3b82f6');

    balls.forEach(b => b.draw());
    redRobots.forEach((r, idx) => {
        r.draw();
        if (idx === activeRedIndex) {
            ctx.strokeStyle = '#facc15';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(r.x, r.y, r.radius + 5, 0, Math.PI * 2);
            ctx.stroke();
        }
    });
    blueRobots.forEach(r => r.draw());

    if (gameState.phase === 'ENDED') {
        ctx.fillStyle = 'rgba(0, 0, 0, 0.85)';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.fillStyle = '#facc15';
        ctx.font = 'bold 28px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText(`比賽結束 - 【${gameState.winner}】獲勝`, canvas.width / 2, 190);
        ctx.fillStyle = '#f8fafc';
        ctx.font = '16px sans-serif';
        ctx.fillText(gameState.winReason, canvas.width / 2, 230);
        ctx.fillText(`最終得分：紅方 ${redTeam.score} 分  vs  藍方 ${blueTeam.score} 分`, canvas.width / 2, 270);
    }
}

function drawTower(tower, knocked, color) {
    ctx.save();
    ctx.beginPath();
    ctx.arc(tower.x, tower.y, tower.radius, 0, Math.PI * 2);
    ctx.fillStyle = knocked ? '#475569' : color;
    ctx.fill();
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.fillStyle = '#fff';
    ctx.font = '10px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText(knocked ? '已擊倒' : '塔環', tower.x, tower.y + 4);
    ctx.restore();
}

function drawFlag(flag, unlocked, knocked, color) {
    ctx.save();
    ctx.fillStyle = knocked ? '#334155' : (unlocked ? color : '#64748b');
    ctx.fillRect(flag.x, flag.y, flag.w, flag.h);
    ctx.strokeStyle = unlocked ? '#facc15' : '#94a3b8';
    ctx.lineWidth = 2;
    ctx.strokeRect(flag.x, flag.y, flag.w, flag.h);

    ctx.font = '11px sans-serif';
    ctx.fillStyle = '#fff';
    ctx.fillText(knocked ? '倒下' : (unlocked ? '可擊倒' : '未解鎖'), flag.x - 12, flag.y - 6);
    ctx.restore();
}

function gameLoop() {
    update();
    drawField();
    requestAnimationFrame(gameLoop);
}

gameLoop();
